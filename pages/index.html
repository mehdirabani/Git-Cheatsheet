دستورگیت کارایی دستور
git -v نمایش ورژن گیتهاب نصب شده
git config.username انتخاب نام کاربری برای کار
git config.email تنظیم ایمیل برای گیت
git config --list نمایش مشخصات ست شده برای گیت
git init معرفی پروژه به گیت
git status نماش و لاگ وضعیت پروژه و گیت
git add "file & folder Name" معرفی فایل یا فولدر مورد نظر به گیت برای رصد تغییرات
git add . معرفی تمامی فایل ها و فوولدر های پروژه به گیت
git commit -m "Massege" معرفی پروژه به عنوان بخش استیبل به گیت به همراه یاداشت
git commit -a -m "Massege" اضافه کردن فایل های مودی فای به گیت و سریع تر شدن قابلیت کامیت
git log نمایش لیست تغییرات و commit های ثبت شده
git log --oneline نمایش هر لاگ در یک خط ( خلاصه و کاربردی )
git log --stat نمایش تغیرات دقیق لاگ ها که چه کد هایی اضافه شده و چه کد هایی از commit های من حذف شده
git log --graph نمایش لاگ‌ها به صورت گرافیکی
git log --graph --oneline نمایش لاگ ها به صورت گرافیکی اما خلاصه
git log --after="23-10-12" نمایش تغیرات بعد از تاریخ 2023/10/12
git log --before="23-10-12" نمایش تغیرات قبل از تاریخ 2023/10/12
git log --author="Mehdi rabani" نمایش تغیرات و کامییت های توصعه دهنده مهدی ربانی
git commit -am "Masege" دقیقا همانند git commit -a -m "Massege" عمل می‌کنه
git show $$$$$$$ نمایش تغییرات در commit $$$$$$$$ مورد نظر
git config --local alias.lgo "log --oneline" برای مخفف سازی از دستورات استفاده می‌شود به عنوان مثال git log --oneline به صورت git lgo نوشته میشود
git branch برای نمایش تعداد شاخه های موجوددر گیت استفاده می‌شود (همون ranch خودمون)
git branch shopingCard بک برنچ یا یک شاخه به اسم shopingCard ساختیم
git switch shopingCard با اساتفاده از این برنچ ما وارد شاخه shopingCard می شویم
git branch -d shopingCard با استفاده از کد بالا ما شاخه shopingCard رو حذف کردیم
git switch -c likePost ما با استفاده از کد بالا هم یک برنچ به نام likePost ساختیم و هم وارد شاخه اون برنچ شدیم
git branch -m table نام شاخه ما رو به table تغییر داد
